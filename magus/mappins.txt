        const combinacionesEspanolas = new Set([
            'a\u0301', 'e\u0301', 'i\u0301', 'o\u0301', 'u\u0301',
            'A\u0301', 'E\u0301', 'I\u0301', 'O\u0301', 'U\u0301',

        const segmentador = new Intl.Segmenter('es', {
            granularity: 'grapheme'
        });

// Motor experimental B, pasa a definitivo
function convertToUnicode(text, style) {
    const mapping = unicodeMappings[style] || unicodeMappings.cursiva;

    return [...segmentador.segment(text)]
        .map(({ segment }) => {
            const descompuesto = segment.normalize('NFD');
            const caracteres = [...descompuesto];

            // Caracteres sin marcas combinantes
            if (caracteres.length === 1) {
                return mapping[caracteres[0]] || segment;
            }

            // Combinaciones diacríticas admitidas
            if (
                caracteres.length === 2 &&
                combinacionesEspanolas.has(descompuesto)
            ) {
                return (mapping[caracteres[0]] || caracteres[0])
                    + caracteres[1];
            }

            // Cualquier otra combinación se conserva intacta
            return segment;
        })
        .join('');
}

        function validateMarkers(text) {
            const pattern = /<<(\w+):([^>]+)>>/g;
            const validStyles = ['cursiva', 'bold', 'script_bold', 'fraktur'];
            const errors = [];
            let match;

            while ((match = pattern.exec(text)) !== null) {
                const style = match[1];
                if (!validStyles.includes(style)) {
                    errors.push(`Estilo "${style}" no válido`);
                }
            }

            const openBrackets = (text.match(/<<(?!\w+:[^>]+>>)/g) || []).length;
            if (openBrackets > 0) {
                errors.push(`${openBrackets} marcador(es) sin cerrar`);
            }

            return errors;
        }

        function updateValidation() {
            const editor = document.getElementById('textEditor');
            const validationStatus = document.getElementById('validationStatus');
            const errors = validateMarkers(editor.value);

            if (errors.length === 0) {
                validationStatus.className = 'success';
                validationStatus.textContent = '✅ Sintaxis correcta';
            } else {
                validationStatus.className = 'error';
                validationStatus.textContent = '❌ ' + errors[0];
            }
        }

        function countCharactersByStyle(text) {
            const counts = { cursiva: 0, bold: 0, script_bold: 0, fraktur: 0, total: text.length };
            const pattern = /<<(\w+):([^>]+)>>/g;
            let match;

            while ((match = pattern.exec(text)) !== null) {
                const style = match[1];
                const content = match[2];
                if (counts.hasOwnProperty(style)) {
                    counts[style] += content.length;
                }
            }

            return counts;
        }

        function updateStats() {
            const editor = document.getElementById('textEditor');
            const counts = countCharactersByStyle(editor.value);

            document.getElementById('totalChars').textContent = counts.total;
            document.getElementById('cursivaCount').textContent = counts.cursiva;
            document.getElementById('boldCount').textContent = counts.bold;
            document.getElementById('scriptCount').textContent = counts.script_bold;
            document.getElementById('frakturCount').textContent = counts.fraktur;
        }

        function saveToHistory() {
            const editor = document.getElementById('textEditor');
            const currentText = editor.value;

            if (historyIndex < history.length - 1) {
                history = history.slice(0, historyIndex + 1);
            }

            history.push(currentText);

            if (history.length > 20) {
                history.shift();
            } else {
                historyIndex++;
            }

            updateHistoryInfo();
        }

        function updateHistoryInfo() {
            document.getElementById('historyInfo').textContent = 
                `Cambios: ${history.length} | Pos: ${historyIndex + 1}`;
        }

        function undoChange() {
            if (historyIndex > 0) {
                historyIndex--;
                document.getElementById('textEditor').value = history[historyIndex];
                processText();
                updateHistoryInfo();
                showStatus('⬅️ Deshecho');
            }
        }

        function redoChange() {
            if (historyIndex < history.length - 1) {
                historyIndex++;
                document.getElementById('textEditor').value = history[historyIndex];
                processText();
                updateHistoryInfo();
                showStatus('➡️ Rehecho');
            }
        }

        function loadTemplate(templateName) {
            const editor = document.getElementById('textEditor');
            if (templates[templateName]) {
                saveToHistory();
                editor.value = templates[templateName];
                processText();

            }
        }

        function handleInput() {
            if (!isProcessing) {
                saveToHistory();
                updateValidation();
                updateStats();
                processText();
            }
        }


        function processText() {
            isProcessing = true;
            const editor = document.getElementById('textEditor');
            const preview = document.getElementById('previewArea');
            const baseStyle = document.getElementById('baseStyle').value;

            let text = editor.value;
            const pattern = /<<(\w+):([^>]+)>>/g;

            let finalText = '';
            let lastIndex = 0;

            text.replace(pattern, (match, style, content, index) => {
                const beforeText = text.substring(lastIndex, index);
                finalText += convertToUnicode(beforeText, baseStyle);
                finalText += convertToUnicode(content, style);
                lastIndex = index + match.length;
                return match;
            });

                finalText += convertToUnicode(text.substring(lastIndex), baseStyle);
                preview.textContent = finalText || 'Tu texto procesado aparecerá aquí...';

                isProcessing = false;
        }

        function copyResult() {
            const preview = document.getElementById('previewArea');
            const text = preview.textContent;

            if (navigator.clipboard && window.isSecureContext) {
                navigator.clipboard.writeText(text).then(() => {
                    showStatus('✅ Copiado al portapapeles');
                });
            } else {
                const textArea = document.createElement('textarea');
                textArea.value = text;
                document.body.appendChild(textArea);
                textArea.select();
                document.execCommand('copy');
                document.body.removeChild(textArea);
                showStatus('✅ Copiado al portapapeles');
            }
        }

            saveToHistory();
            document.getElementById('textEditor').value = '';
